-- ----------------------------------------------------------------------------
-- 1. EXTENSIONES
-- ----------------------------------------------------------------------------
create extension if not exists pgcrypto;

-- ----------------------------------------------------------------------------
-- 2. TABLA: accounts
--    Almacena cada cuenta de usuario registrada (para el ranking público)
-- ----------------------------------------------------------------------------
create table if not exists public.accounts (
  id uuid primary key default gen_random_uuid(),
  username text unique not null,
  created_at timestamptz not null default now(),
  profile_photo text,          -- URL de la foto de perfil (data URL o externa)
  description text,            -- Descripción de la cuenta
  downloads integer not null default 0,   -- Total de descargas realizadas
  redeemed_codes text[] not null default '{}',   -- Códigos canjeados
  skip_until timestamptz       -- Expiración del acceso directo (skip de pasos)
);

-- ----------------------------------------------------------------------------
-- 3. TABLA: download_events
--    Registro de cada descarga individual (opcional, para historial)
-- ----------------------------------------------------------------------------
create table if not exists public.download_events (
  id bigint generated by default as identity primary key,
  account_id uuid not null references public.accounts(id) on delete cascade,
  app_slug text not null,      -- Slug de la app descargada
  created_at timestamptz not null default now()
);

-- ----------------------------------------------------------------------------
-- 4. ÍNDICES
-- ----------------------------------------------------------------------------
create index if not exists accounts_username_idx on public.accounts (username);
create index if not exists accounts_downloads_idx on public.accounts (downloads desc);
create index if not exists download_events_account_idx on public.download_events (account_id);
create index if not exists download_events_created_idx on public.download_events (created_at desc);

-- ----------------------------------------------------------------------------
-- 5. ROW LEVEL SECURITY (RLS)
-- ----------------------------------------------------------------------------
alter table public.accounts enable row level security;
alter table public.download_events enable row level security;

-- Cualquiera puede leer las cuentas (necesario para el ranking público)
create policy if not exists accounts_select_all
  on public.accounts for select using (true);

-- Cualquiera puede leer los eventos de descarga
create policy if not exists download_events_select_all
  on public.download_events for select using (true);

-- Las inserciones y actualizaciones se realizan desde las API routes
-- del sitio usando la service_role key (NO se exponen al público).

-- ----------------------------------------------------------------------------
-- 6. COMENTARIOS
-- ----------------------------------------------------------------------------
comment on table public.accounts is 'Cuentas de usuarios registrados (ranking público)';
comment on table public.download_events is 'Historial de descargas por cuenta';

- El ranking lee de la tabla public.accounts ordenado por downloads desc.
